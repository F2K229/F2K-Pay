// Initier un paiement F2KPay
const response = await fetch('https://f2kpay-backend.onrender.com/api/payments/initiate', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'X-F2K-Public-Key': 'pk_live_VOTRE_CLE_PUBLIQUE'
  },
  body: JSON.stringify({
    slug: 'votre-lien-slug',
    customer: {
      firstName: 'Jean',
      lastName: 'Dupont',
      email: 'jean@email.com',
      phone: '22997000000'
    }
  })
});

const data = await response.json();
if (data.success) {
  // Rediriger vers la page de paiement
  window.location.href = data.data.checkoutUrl;
}

▶ Initier un paiement
<?php
$curl = curl_init();
curl_setopt_array($curl, [
  CURLOPT_URL => 'https://f2kpay-backend.onrender.com/api/payments/initiate',
  CURLOPT_RETURNTRANSFER => true,
  CURLOPT_POST => true,
  CURLOPT_HTTPHEADER => [
    'Content-Type: application/json',
    'X-F2K-Public-Key: pk_live_VOTRE_CLE_PUBLIQUE'
  ],
  CURLOPT_POSTFIELDS => json_encode([
    'slug' => 'votre-lien-slug',
    'customer' => [
      'firstName' => 'Jean',
      'lastName'  => 'Dupont',
      'email'     => 'jean@email.com',
      'phone'     => '22997000000'
    ]
  ])
]);
$response = json_decode(curl_exec($curl), true);
if ($response['success']) {
  header('Location: ' . $response['data']['checkoutUrl']);
}
?>
